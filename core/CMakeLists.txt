cmake_minimum_required(VERSION 3.16)
project(tetrachords_core VERSION 0.1.0 LANGUAGES CXX)

add_library(tetrachords_core STATIC src/engine.cpp)
target_include_directories(tetrachords_core PUBLIC include)
target_compile_features(tetrachords_core PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tetrachords_core PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti -ffp-contract=off)
endif()

option(TETRACHORDS_BUILD_TESTS "Build host-only test executables" ON)
if(TETRACHORDS_BUILD_TESTS)
    enable_testing()
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE tetrachords_core)
    add_executable(parity_driver tests/parity_driver.cpp)
    target_link_libraries(parity_driver PRIVATE tetrachords_core)
    add_executable(timing_tests tests/timing_tests.cpp)
    target_link_libraries(timing_tests PRIVATE tetrachords_core)
    add_executable(timing_driver tests/timing_driver.cpp)
    target_link_libraries(timing_driver PRIVATE tetrachords_core)
    # Assertions are required even in a Release host-test build.
    if(MSVC)
        target_compile_options(core_tests PRIVATE /UNDEBUG)
        target_compile_options(timing_tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(core_tests PRIVATE -UNDEBUG)
        target_compile_options(timing_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME core_tests COMMAND core_tests)
    add_test(NAME timing_tests COMMAND timing_tests)
    find_program(NODE_EXECUTABLE node)
    if(NODE_EXECUTABLE)
        add_test(NAME actual_js_parity COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/parity.js --driver $<TARGET_FILE:parity_driver>)
        add_test(NAME actual_js_timing_parity COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/timing_parity.js --driver $<TARGET_FILE:timing_driver>)
    endif()
endif()
